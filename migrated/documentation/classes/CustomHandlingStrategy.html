/**
 * Hybrid mode
 */
export class CustomHandlingStrategy implements UrlHandlingStrategy {
  shouldProcessUrl(url) {
    return url.toString().startsWith("/ng2") || url.toString() == "/"
  }
  extract(url) { return url }
  merge(url, whole) { return url }
}
